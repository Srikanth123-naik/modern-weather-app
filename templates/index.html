import cv2 import face_recognition # Load known image known_image = face_recognition.load_image_file("myface.jpg") known_encoding = face_recognition.face_encodings(known_image)[0] # Open camera cam = cv2.VideoCapture(0) while True: ret, frame = cam.read() if not ret: print("Camera not working") break # Resize for speed small_frame = cv2.resize(frame, (0, 0), fx=0.25, fy=0.25) # Convert BGR to RGB rgb_small_frame = cv2.cvtColor(small_frame, cv2.COLOR_BGR2RGB) # Detect faces face_locations = face_recognition.face_locations(rgb_small_frame) # Encode faces face_encodings = face_recognition.face_encodings( rgb_small_frame, face_locations ) # Loop through faces for (top, right, bottom, left), face_encoding in zip( face_locations, face_encodings ): # Compare face matches = face_recognition.compare_faces( [known_encoding], face_encoding ) name = "Unknown" if True in matches: name = "Srikanth" # Scale back top *= 4 right *= 4 bottom *= 4 left *= 4 # Draw rectangle cv2.rectangle( frame, (left, top), (right, bottom), (0, 255, 0), 2 ) # Put name cv2.putText( frame, name, (left, top - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 255, 0), 2 ) cv2.imshow("AI Face Recognition", frame) # Press q to exit if cv2.waitKey(1) & 0xFF == ord('q'): break cam.release() cv2.destroyAllWindows()
